// 指数为高精度时 输入为字符串
vector<int> polypow(const vector<int>& f, string p, int sz) {
    vector<int> ans = f;
    ans.resize(sz);
    int a = 1, b = 0, k1 = 0, k2 = 0;//a * x ^ b
    while (b < sz && !ans[b]) ++b;
    for (auto ch : p) {
        k1 = k1 * 10 % P;
        k1 = (k1 + ch - '0') % P;
        k2 = k2 * 10 % (P - 1);
        k2 = (k2 + ch - '0') % (P - 1);
        if (b && k1 * b >= sz) {
            fill(all(ans), 0);
            return ans;
        }
    }
    if (b) {
        for (int i = 0;i + b < sz;++i) ans[i] = ans[i + b];
        for (int i = sz - b;i < sz;++i) ans[i] = 0;
    }
    if (ans[0] != 1) {
        a = qpow(ans[0], k2);
        int inva = inv(ans[0]);
        for (int i = 0;i < sz;++i) (ans[i] *= inva) %= P;
    }
    b *= k1;
    ans = polyln(ans);
    for (auto& v : ans) (v *= k1) %= P;
    ans = polyexp(ans);
    if (b) {
        for (int i = sz;i >= b;--i)ans[i] = ans[i - b];
        for (int i = 0;i < min(b, sz);++i) ans[i] = 0;
    }
    if (a != 1)
        for (int i = b;i < sz;++i) (ans[i] *= a) %= P;
    return ans;
}
