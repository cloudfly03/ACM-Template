#define complex complex<ld>
#define PI (ld)(3.141592653589793)

void FFT(vector<complex>& A, bool rev) {
    auto f = [](auto f, vector<complex>& A, bool rev) {
        int n = A.size();
        if (n == 1) return;
        vector<complex> A1(n / 2), A2(n / 2);
        for (int i = 0;i < n / 2;++i)
            A1[i] = A[i * 2], A2[i] = A[i * 2 + 1];
        f(f, A1, rev);f(f, A2, rev);
        complex w1(cos(2 * PI / n), (rev ? -1 : 1) * sin(2 * PI / n));
        complex wk(1, 0);
        for (int i = 0;i < n / 2;++i) {
            A[i] = A1[i] + A2[i] * wk;
            A[i + n / 2] = A1[i] - A2[i] * wk;
            wk = wk * w1;
        }
        };
    f(f, A, rev);
    if (!rev) return;
    for (auto& v : A) v /= n;
}
