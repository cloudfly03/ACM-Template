#define P (int)(998244353)
#define G (int)(3)

int qpow(int base, int p) {
    int ans = 1;
    while (p) {
        if (p & 1) ans = ans * base % P;
        base = base * base % P;
        p >>= 1;
    }
    return ans;
}

int inv(int x) {
    return qpow(x, P - 2);
}

void NTT(vector<int>& A, bool rev) {
    auto f = [](auto f, vector<int>& A, bool rev) {
        int n = A.size();
        if (n == 1) return;
        vector<int> A1(n / 2), A2(n / 2);
        for (int i = 0;i < n / 2;++i)
            A1[i] = A[i * 2], A2[i] = A[i * 2 + 1];
        f(f, A1, rev);f(f, A2, rev);
        int g1 = qpow((rev ? inv(G) : G), (P - 1) / A.size());
        int gk = 1;
        for (int i = 0;i < n / 2;++i) {
            A[i] = (A1[i] + A2[i] * gk % P) % P;
            A[i + n / 2] = ((A1[i] - A2[i] * gk % P) % P + P) % P;
            gk = gk * g1 % P;
        }
    };
    f(f, A, rev);
    if (!rev) return;
    int ni = inv(A.size());
    for (auto& v : A) (v *= ni) %= P;
}
