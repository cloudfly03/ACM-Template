vector<int> polyexp(const vector<int>& f) {
    int n = 1, ni, sz = f.size();
    while (n < sz) n <<= 1;
    vector<int> ans{ 1 };
    for (int m = 2; m <= n;m <<= 1) {
        vector<int> a(m << 1);
        ans.resize(m << 1, 0);
        for (int i = 0;i < min(m, sz);++i) a[i] = f[i];
        vector<int> b = polyln(ans);
        for (int i = 0;i < m << 1;++i) b[i] = ((-b[i] + a[i]) % P + P) % P;
        (b[0] += 1) %= P;
        ans = polymul(ans, b);
        ans.resize(m);
    }
    ans.resize(sz);
    return ans;
}
